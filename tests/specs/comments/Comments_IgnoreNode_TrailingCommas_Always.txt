~~ trailingCommas: always ~~
== should not add a trailing comma to an ignored node ==
{
  "tasks": {
    // dprint-ignore
    "foo": "echo bar"
  },
  "other": {
    "foo":   "echo bar"
  },
  "array": [
    1,
    // dprint-ignore
    [ 2,   3 ]
  ]
}

[expect]
{
  "tasks": {
    // dprint-ignore
    "foo": "echo bar"
  },
  "other": {
    "foo": "echo bar",
  },
  "array": [
    1,
    // dprint-ignore
    [ 2,   3 ]
  ],
}
