== should write the top level fields in the conventional order ==
{
  "devDependencies": {
    "typescript": "^5.0.0",
    "@types/node": "^20.0.0"
  },
  "scripts": {
    "test": "node --test",
    "build": "tsc"
  },
  "version": "1.0.0",
  "dependencies": {
    "zod": "^3.0.0",
    "chalk": "^5.0.0"
  },
  "name": "my-package",
  "license": "MIT"
}

[expect]
{
  "name": "my-package",
  "version": "1.0.0",
  "license": "MIT",
  "scripts": {
    "test": "node --test",
    "build": "tsc"
  },
  "dependencies": {
    "chalk": "^5.0.0",
    "zod": "^3.0.0"
  },
  "devDependencies": {
    "@types/node": "^20.0.0",
    "typescript": "^5.0.0"
  }
}
