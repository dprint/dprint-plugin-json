== should write the top level fields in the conventional order ==
{
  "devDependencies": {
    "typescript": "^5.0.0",
    "@types/node": "^20.0.0"
  },
  "scripts": {
    "test": "node --test",
    "build": "tsc"
  },
  "version": "1.0.0",
  "dependencies": {
    "zod": "^3.0.0",
    "chalk": "^5.0.0"
  },
  "name": "my-package",
  "license": "MIT"
}

[expect]
{
  "name": "my-package",
  "version": "1.0.0",
  "license": "MIT",
  "scripts": {
    "test": "node --test",
    "build": "tsc"
  },
  "dependencies": {
    "chalk": "^5.0.0",
    "zod": "^3.0.0"
  },
  "devDependencies": {
    "@types/node": "^20.0.0",
    "typescript": "^5.0.0"
  }
}

== should sort unknown fields after the known ones and underscore fields last ==
{
  "_id": "my-package@1.0.0",
  "zzz": true,
  "dependencies": {},
  "_from": "my-package",
  "aaa": true,
  "name": "my-package"
}

[expect]
{
  "name": "my-package",
  "dependencies": {},
  "aaa": true,
  "zzz": true,
  "_from": "my-package",
  "_id": "my-package@1.0.0"
}

== should place a section by convention and sort its contents ==
{
  "overrides": { "b": "1", "a": "1" },
  "resolutions": { "b": "1", "a": "1" },
  "engines": { "npm": "1", "node": "1" },
  "bin": { "b": "./b.js", "a": "./a.js" },
  "name": "my-package"
}

  "name": "my-package",
  "bin": { "a": "./a.js", "b": "./b.js" },

  "overrides": { "a": "1", "b": "1" },
  "engines": { "node": "1", "npm": "1" }
}
