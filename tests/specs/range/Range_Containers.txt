== should format the whole object when the range reaches its brackets ==
{
  "a":1,
  "b":[|{"c":1,
"d":2}|]
}

[expect]
{
  "a":1,
  [|"b": { "c": 1, "d": 2 }|]
}

== should format the whole object when the line breaks around the members change ==
{
  "a":1,
  "b":{
"c":[||]1, "d":2}
}

[expect]
{
  "a":1,
  "b":{
    "c": 1,
    "d": 2
  }
}

== should format the file when the range reaches the root brackets ==
[|{"a":1,
"b":2}|]

[expect]
{ "a": 1, "b": 2 }

== should format the file when the range starts at the root bracket ==
[|{"a"|]:1,
"b":2}

[expect]
{ "a": 1, "b": 2 }
