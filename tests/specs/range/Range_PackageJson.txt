-- package.json --
== should not move a touched property that the conventions would move ==
{
  [|"version":"1.0.0"|],
  "dependencies": {"b":"1",
    "a":"1"},
  "name":"a"
}

[expect]
{
  [|"version": "1.0.0",|]
  "dependencies": {"b":"1",
    "a":"1"},
  "name":"a"
}

== should not reorder the touched top level properties ==
{
  [|"version":"1.0.0",
  "name":"a"|],
  "license":"MIT"
}

[expect]
{
  [|"version": "1.0.0",
  "name": "a",|]
  "license":"MIT"
}

== should not reorder the properties of a section when one is not touched ==
{
  "name":"a",
  "dependencies": {
    [|"c":"1",
    "b":"1"|],
    "a":"1"
  }
}

[expect]
{
  "name":"a",
  "dependencies": {
    [|"c": "1",
    "b": "1",|]
    "a":"1"
  }
}

== should not reorder the properties of a section when only one is touched ==
{
  "name":"a",
  "dependencies": {
    "c":"1",
    [|"b":"1"|],
    "a":"1"
  }
}

[expect]
{
  "name":"a",
  "dependencies": {
    "c":"1",
    [|"b": "1",|]
    "a":"1"
  }
}

== should reorder the properties of a section when all of them are touched ==
{
  "version":"1.0.0",
  "dependencies": {
    [|"c":"1",
    "b":"1",
    "a":"1"|]
  },
  "name":"a"
}

[expect]
{
  "version":"1.0.0",
  "dependencies": {
    [|"a": "1",
    "b": "1",
    "c": "1"|]
  },
  "name":"a"
}

== should reorder the properties of a section when the range is from its first to its last ==
{
  "version":"1.0.0",
  "dependencies": {
    "c"[|:"1",
    "b":"1",
    "a"|]:"1"
  },
  "name":"a"
}

[expect]
{
  "version":"1.0.0",
  "dependencies": {
    "a": "1",
    "b": "1",
    "c": "1"
  },
  "name":"a"
}

== should format the member holding a section when reordering all of its properties changes its line breaks ==
{
  "name":"a",
  "version":"1.0.0",
  "dependencies": {[|"b":"1",
    "a"|]:"1"}
}

[expect]
{
  "name":"a",
  "version":"1.0.0",
  "dependencies": { "a": "1", "b": "1" }
}

== should not move the member holding a section that is formatted ==
{
  "version":"1.0.0",
  "dependencies": {[|"b":"1",
    "a"|]:"1"},
  "name":"a"
}

[expect]
{
  "version":"1.0.0",
  "dependencies": { "a": "1", "b": "1" },
  "name":"a"
}
